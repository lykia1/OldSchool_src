	ALTER PROCEDURE [dbo].[atum_Market_Get_Enchant_Info]
		@i_ItemUID					bigint	
	AS
	SELECT 
    EnchantItemNum, 
    COUNT(EnchantItemNum)
FROM
    td_enchant
	where TargetItemUniqueNumber =@i_ItemUID
GROUP BY EnchantItemNum
